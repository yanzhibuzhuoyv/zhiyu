cmake_minimum_required(VERSION 3.22)
project(zhiyu_qwen_edge_diffusion LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(qwen-edge-condition-test tests/condition_test.cpp)
target_include_directories(qwen-edge-condition-test PRIVATE include)
target_compile_options(qwen-edge-condition-test PRIVATE -Wall -Wextra -Werror)
add_test(NAME qwen-edge-condition-contract COMMAND qwen-edge-condition-test)
add_executable(qwen-edge-vae-probe-test tests/vae_probe_test.cpp)
target_include_directories(qwen-edge-vae-probe-test PRIVATE include)
target_compile_options(qwen-edge-vae-probe-test PRIVATE -Wall -Wextra -Werror)
add_test(NAME qwen-edge-vae-probe-contract COMMAND qwen-edge-vae-probe-test)
add_executable(qwen-edge-latent-checkpoint-test tests/latent_checkpoint_test.cpp)
target_include_directories(qwen-edge-latent-checkpoint-test PRIVATE include)
target_compile_options(qwen-edge-latent-checkpoint-test PRIVATE -Wall -Wextra -Werror)
add_test(NAME qwen-edge-latent-checkpoint-contract COMMAND qwen-edge-latent-checkpoint-test)

option(QWEN_EDGE_BUILD_RUNTIME "Build the pinned, patched stable-diffusion.cpp" OFF)
if(QWEN_EDGE_BUILD_RUNTIME)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    execute_process(COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../../scripts/qwen_diffusion_prepare.py"
        --verify-only --source "${QWEN_EDGE_SD_SOURCE}" COMMAND_ERROR_IS_FATAL ANY)
    set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SD_WEBP OFF CACHE BOOL "" FORCE)
    set(SD_WEBM OFF CACHE BOOL "" FORCE)
    # Bundled Windows CMake with MinGW can leave the processor blank and select
    # generic ggml kernels. Resolve only that missing host value, never Android.
    if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND NOT CMAKE_SYSTEM_PROCESSOR)
        execute_process(COMMAND "${CMAKE_C_COMPILER}" -dumpmachine
            OUTPUT_VARIABLE COMPILER_TARGET OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
        if(COMPILER_TARGET MATCHES "^x86_64-")
            set(CMAKE_SYSTEM_PROCESSOR "AMD64")
        endif()
    endif()
    add_subdirectory("${QWEN_EDGE_SD_SOURCE}" sd-runtime)
    add_executable(qwen-edge-smoke smoke.cpp)
    target_include_directories(qwen-edge-smoke PRIVATE include)
    target_link_libraries(qwen-edge-smoke PRIVATE stable-diffusion)
    add_executable(qwen-edge-vae-probe vae_probe.cpp)
    target_include_directories(qwen-edge-vae-probe PRIVATE include)
    # The pinned fork compiles ggml tensor metadata with this ABI layout.
    target_compile_definitions(qwen-edge-vae-probe PRIVATE GGML_MAX_NAME=160)
    target_link_libraries(qwen-edge-vae-probe PRIVATE stable-diffusion)
    # Same VAE runtime implementation, different explicit input contract.
    add_executable(qwen-edge-latent-decode vae_probe.cpp)
    target_include_directories(qwen-edge-latent-decode PRIVATE include)
    target_compile_definitions(qwen-edge-latent-decode PRIVATE GGML_MAX_NAME=160 QWEN_EDGE_CHECKPOINT_DECODE=1)
    target_link_libraries(qwen-edge-latent-decode PRIVATE stable-diffusion)
    if(WIN32)
        target_link_libraries(qwen-edge-vae-probe PRIVATE psapi)
        target_link_libraries(qwen-edge-latent-decode PRIVATE psapi)
    endif()
    add_executable(qwen-edge-vae-structure-test tests/vae_structure_test.cpp)
    target_include_directories(qwen-edge-vae-structure-test PRIVATE include)
    target_compile_definitions(qwen-edge-vae-structure-test PRIVATE GGML_MAX_NAME=160)
    target_link_libraries(qwen-edge-vae-structure-test PRIVATE stable-diffusion)
    add_test(NAME qwen-edge-vae-synthetic-metadata COMMAND qwen-edge-vae-structure-test)
endif()
