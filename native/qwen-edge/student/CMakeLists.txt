cmake_minimum_required(VERSION 3.22)
project(qwen_student_features LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(QWEN_LLAMA_COMMIT "23b0202a189c44a54625aadcb37a946dd1d6278d")
set(LLAMA_SOURCE_DIR "" CACHE PATH "Separate, unchanged upstream llama.cpp checkout")
if(NOT EXISTS "${LLAMA_SOURCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Pass -DLLAMA_SOURCE_DIR=<locked upstream checkout>")
endif()
execute_process(COMMAND git -C "${LLAMA_SOURCE_DIR}" rev-parse HEAD
  OUTPUT_VARIABLE ACTUAL_LLAMA_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE
  COMMAND_ERROR_IS_FATAL ANY)
if(NOT ACTUAL_LLAMA_COMMIT STREQUAL QWEN_LLAMA_COMMIT)
  message(FATAL_ERROR "llama.cpp must be locked to ${QWEN_LLAMA_COMMIT}")
endif()
execute_process(COMMAND git -C "${LLAMA_SOURCE_DIR}" status --porcelain
  OUTPUT_VARIABLE LLAMA_DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE
  COMMAND_ERROR_IS_FATAL ANY)
if(LLAMA_DIRTY)
  message(FATAL_ERROR "Upstream llama.cpp checkout must have no local edits")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
# Android's bundled Windows CMake leaves this blank with MinGW. Identify the
# compiler target instead of silently using GGML's very slow generic CPU path.
if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND NOT CMAKE_SYSTEM_PROCESSOR)
  execute_process(COMMAND "${CMAKE_C_COMPILER}" -dumpmachine
    OUTPUT_VARIABLE COMPILER_TARGET OUTPUT_STRIP_TRAILING_WHITESPACE
    COMMAND_ERROR_IS_FATAL ANY)
  if(COMPILER_TARGET MATCHES "^x86_64-")
    set(CMAKE_SYSTEM_PROCESSOR "AMD64")
  endif()
endif()
add_subdirectory("${LLAMA_SOURCE_DIR}" llama)
add_library(qwen-student-runtime STATIC student_runtime.cpp)
target_link_libraries(qwen-student-runtime PUBLIC llama)
target_include_directories(qwen-student-runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(qwen-student-runtime PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_executable(qwen-student-extract main.cpp)
add_executable(qwen-student-tokenize tokenize_cli.cpp)
target_link_libraries(qwen-student-tokenize PRIVATE qwen-student-runtime)
add_executable(qwen-teacher-quantize teacher_quantize.cpp)
target_link_libraries(qwen-teacher-quantize PRIVATE llama)
add_executable(qwen-teacher-extract teacher_extract.cpp)
target_link_libraries(qwen-teacher-extract PRIVATE llama)
target_compile_features(qwen-teacher-extract PRIVATE cxx_std_20)
add_executable(teacher-contract-test teacher_contract_test.cpp)
target_compile_features(teacher-contract-test PRIVATE cxx_std_20)
target_link_libraries(qwen-student-extract PRIVATE qwen-student-runtime vendor::hash)
target_compile_definitions(qwen-student-extract PRIVATE QWEN_LLAMA_COMMIT="${QWEN_LLAMA_COMMIT}")
if(WIN32)
  target_link_libraries(qwen-student-runtime PRIVATE psapi)
endif()
add_executable(feature-contract-test feature_contract_test.cpp)
add_executable(student-runtime-test student_runtime_test.cpp)
target_link_libraries(student-runtime-test PRIVATE qwen-student-runtime)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(qwen-student-extract PRIVATE -Wall -Wextra -Werror)
  target_compile_options(qwen-student-runtime PRIVATE -Wall -Wextra -Werror)
  target_compile_options(feature-contract-test PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME feature-contract COMMAND feature-contract-test)
add_test(NAME student-runtime-contract COMMAND student-runtime-test)
add_test(NAME teacher-contract COMMAND teacher-contract-test)
