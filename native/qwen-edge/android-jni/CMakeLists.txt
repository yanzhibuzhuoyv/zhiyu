cmake_minimum_required(VERSION 3.22)
project(zhiyu_qwen_android_jni LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(ZHIYU_QWEN_JNI_MODE "" CACHE STRING "TEXT or IMAGE; separate builds keep two GGML revisions separate")
if(NOT ANDROID)
    message(FATAL_ERROR "JNI runtime builds require the Android NDK toolchain")
endif()
if(NOT ANDROID_ABI MATCHES "^(arm64-v8a|x86_64)$")
    message(FATAL_ERROR "Only arm64-v8a and x86_64 are approved for this JNI runtime")
endif()
get_filename_component(QWEN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../diffusion/upstream-lock.json" SD_LOCK)
string(JSON SD_COMMIT GET "${SD_LOCK}" commit)
string(JSON SD_PATCH GET "${SD_LOCK}" patch_sha256)
string(JSON SD_GGML GET "${SD_LOCK}" submodules ggml)
string(JSON SD_FORK GET "${SD_LOCK}" fork_revision)
set(IDENTITY "qwen-edge-android-jni-v1|llama=23b0202a189c44a54625aadcb37a946dd1d6278d|sd=${SD_COMMIT}|ggml=${SD_GGML}|fork=${SD_FORK}|patch=${SD_PATCH}")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_KOMPUTE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)

if(ZHIYU_QWEN_JNI_MODE STREQUAL "TEXT")
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../student" student-runtime)
    add_library(zhiyu-qwen-adapter-jni STATIC "${CMAKE_CURRENT_SOURCE_DIR}/../adapter/adapter.cpp")
    target_compile_features(zhiyu-qwen-adapter-jni PRIVATE cxx_std_20)
    target_include_directories(zhiyu-qwen-adapter-jni PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../adapter")
    set(JNI_TARGET zhiyu_qwen_text_jni)
    add_library(${JNI_TARGET} SHARED text_jni.cpp)
    target_link_libraries(${JNI_TARGET} PRIVATE qwen-student-runtime zhiyu-qwen-adapter-jni log)
    set(EXPORTS "${CMAKE_CURRENT_SOURCE_DIR}/text.exports.map")
elseif(ZHIYU_QWEN_JNI_MODE STREQUAL "IMAGE")
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${QWEN_ROOT}/scripts/qwen_diffusion_prepare.py"
        --source "${QWEN_EDGE_SD_SOURCE}" --verify-only COMMAND_ERROR_IS_FATAL ANY)
    set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SD_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(SD_BUILD_SHARED_GGML_LIB OFF CACHE BOOL "" FORCE)
    set(SD_WEBP OFF CACHE BOOL "" FORCE)
    set(SD_WEBM OFF CACHE BOOL "" FORCE)
    add_subdirectory("${QWEN_EDGE_SD_SOURCE}" image-runtime)
    set(JNI_TARGET zhiyu_qwen_image_jni)
    add_library(${JNI_TARGET} SHARED image_jni.cpp sample_reuse.cpp decode_reuse.cpp)
    target_compile_definitions(${JNI_TARGET} PRIVATE GGML_MAX_NAME=160)
    target_link_libraries(${JNI_TARGET} PRIVATE stable-diffusion log)
    set(EXPORTS "${CMAKE_CURRENT_SOURCE_DIR}/image.exports.map")
else()
    message(FATAL_ERROR "Pass -DZHIYU_QWEN_JNI_MODE=TEXT or IMAGE")
endif()
target_include_directories(${JNI_TARGET} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../diffusion/include")
target_compile_features(${JNI_TARGET} PRIVATE cxx_std_20)
target_compile_definitions(${JNI_TARGET} PRIVATE ZHIYU_QWEN_RUNTIME_IDENTITY="${IDENTITY}")
target_compile_options(${JNI_TARGET} PRIVATE -Wall -Wextra)
target_link_options(${JNI_TARGET} PRIVATE "-Wl,--version-script=${EXPORTS}" "-Wl,--exclude-libs,ALL"
    -Wl,--no-undefined -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
set_target_properties(${JNI_TARGET} PROPERTIES OUTPUT_NAME ${JNI_TARGET}
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/jniLibs" LINK_DEPENDS "${EXPORTS}")
