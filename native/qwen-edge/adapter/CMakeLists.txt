cmake_minimum_required(VERSION 3.22)
project(zhiyu_qwen_adapter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(zhiyu_qwen_adapter STATIC adapter.cpp)
target_include_directories(zhiyu_qwen_adapter PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(zhiyu_qwen_adapter PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_executable(qwen-adapter-project adapter_cli.cpp)
target_link_libraries(qwen-adapter-project PRIVATE zhiyu_qwen_adapter)
option(ZHIYU_QWEN_ADAPTER_TESTS "Build isolated adapter contract tests" ON)
if(ZHIYU_QWEN_ADAPTER_TESTS)
    enable_testing()
    add_executable(qwen_adapter_test adapter_test.cpp)
    target_link_libraries(qwen_adapter_test PRIVATE zhiyu_qwen_adapter)
    add_test(NAME adapter_contract COMMAND qwen_adapter_test)
endif()
